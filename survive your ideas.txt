local settings = { values = {}, toggles = {} }
local settingsLoaded = false

local function loadSettings()
    if not (isfile and readfile) then return end
    local ok, data = pcall(function()
        if isfile(SETTINGS_FILE) then
            return HttpService:JSONDecode(readfile(SETTINGS_FILE))
        end
    end)
    if ok and type(data) == "table" then
        settings.values = type(data.values) == "table" and data.values or {}
        settings.toggles = type(data.toggles) == "table" and data.toggles or {}
        settingsLoaded = true
    end
end

local saveQueued = false
local function saveSettings()
    if saveQueued or not writefile then return end
    saveQueued = true
    task.delay(0.3, function()
        saveQueued = false
        pcall(function()
            writefile(SETTINGS_FILE, HttpService:JSONEncode(settings))
        end)
    end)
end

loadSettings()

for _, t in ipairs(THEMES) do
    if settings.values.theme == t.name then accent = t.color end
end

local pages = {}

local function selectTab(name)
    activeTab = name
    for n, page in pairs(pages) do
        page.Visible = (n == name)
    end
    for n, t in pairs(tabButtons) do
        local on = (n == name)
        tween(t.button, { BackgroundColor3 = on and C.row or C.side })
        tween(t.label, { TextColor3 = on and C.text or C.sub })
        t.bar.Visible = on
    end
end

local function createBox(row, key, default, placeholder)
    local box = Instance.new("TextBox")
    box.Parent = row
    box.ZIndex = 3
    box.BackgroundColor3 = C.input
    box.BorderSizePixel = 0
    box.Size = UDim2.new(0, 70, 0, 24)
    box.ClearTextOnFocus = false
    box.Font = Enum.Font.GothamBold
    box.PlaceholderText = placeholder or ""
    box.PlaceholderColor3 = C.sub
    box.Text = tostring(settings.values[key] ~= nil and settings.values[key] or default)
    box.TextColor3 = C.text
    box.TextSize = 12
    corner(box, 6)

    local bs = Instance.new("UIStroke")
    bs.Thickness = 1
    bs.Transparency = 1
    bs.Parent = box
    onAccent(function(c) bs.Color = c end)
    track(box.Focused:Connect(function() tween(bs, { Transparency = 0 }) end))

    track(box.FocusLost:Connect(function()
        tween(bs, { Transparency = 1 })
        if numericKeys[key] and not tonumber(box.Text) then
            box.Text = tostring(default)
        end
        settings.values[key] = box.Text
        saveSettings()
    end))

    boxes[key] = box
    settings.values[key] = box.Text
    return box
end

local function addActionRow(label, kind, callback, boxKey, boxDefault, placeholder)
    local row = newRow()
    local btn = Instance.new("TextButton")
    btn.Parent = row
    btn.AutoButtonColor = false
    btn.BorderSizePixel = 0
    btn.Font = Enum.Font.GothamBold
    btn.Text = label
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.TextSize = 12
    corner(btn, 8)

    local function baseColor()
        if kind == "danger" then return C.danger end
        if kind == "neutral" then return C.off end
        return accent
    end
    onAccent(function() btn.BackgroundColor3 = baseColor() end)

    if boxKey then
        btn.Size = UDim2.new(1, -88, 1, 0)
        local box = createBox(row, boxKey, boxDefault, placeholder)
        box.Position = UDim2.new(1, -78, 0.5, -12)
    else
        btn.Size = UDim2.new(1, 0, 1, 0)
    end

    btn.MouseEnter:Connect(function()
        tween(btn, { BackgroundColor3 = baseColor():Lerp(Color3.new(1, 1, 1), 0.15) })
    end)
    btn.MouseLeave:Connect(function() tween(btn, { BackgroundColor3 = baseColor() }) end)
    track(btn.MouseButton1Click:Connect(callback))
    return btn
end

    local function parseFall(model)
        local raw = model:GetAttribute("AirDropFall")
        if type(raw) ~= "string" then return nil end
        local ok, d = pcall(HttpService.JSONDecode, HttpService, raw)
        if not ok or type(d) ~= "table" or type(d.land) ~= "table" then return nil end
        return Vector3.new(d.land[1], d.land[2], d.land[3]), tonumber(d.start), tonumber(d.secs) or 2.9

        local t0 = tick()

            task.wait(0.05)

        task.wait(0.4)

        if parent:IsA("Model") and parent:GetAttribute("AirDropFall") then
            table.insert(candidates, parent)
        else
            for _, m in ipairs(parent:GetChildren()) do
                if m:IsA("Model") and (m.Name == "AirDrop" or m:GetAttribute("AirDropFall")) then
                    table.insert(candidates, m)
                end
            end
        end

        for _, m in ipairs(candidates) do
            local land, start, secs = waitFall(m)
            if land and start then
                local dist = (land - hrp.Position).Magnitude
                if not bestDist or dist < bestDist then
                    bestModel, bestLand, bestStart, bestSecs, bestDist = m, land, start, secs, dist
                end
            end
        end

        if bestDist > getNum("collectRadius", 250) then return end

        landingAt[kind] = tick()

        local wait_t = (bestStart + bestSecs) - serverNow() - 0.05
        wait_t = math.clamp(wait_t, 0, bestSecs + 0.5)
        if wait_t > 0 then task.wait(wait_t) end

    local function findCover()
        local hrp = getHRP()
        if not hrp then return nil end
        local center = hrp.Position

        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.IgnoreWater = true
        local ignore = {}
        for _, p in ipairs(Players:GetPlayers()) do
            if p.Character then table.insert(ignore, p.Character) end
        end
        if safePlatform then table.insert(ignore, safePlatform) end
        params.FilterDescendantsInstances = ignore

        local dirs = {}
        for i = 0, 7 do
            local a = math.rad(i * 45)
            table.insert(dirs, Vector3.new(math.cos(a), 0, math.sin(a)))
        end
        local cardinals = { dirs[1], dirs[3], dirs[5], dirs[7] }
        local MAXD = 30
        local best, bestScore, bestDist, bestWalls = nil, -math.huge, math.huge, 0
        local step, range, n = 8, 120, 0

        local function wallDist(x, ground, z, dir)
            local low = workspace:Raycast(Vector3.new(x, ground + 2, z), dir * MAXD, params)
            if not (low and low.Instance.CanCollide) then return nil end
            local high = workspace:Raycast(Vector3.new(x, ground + 7, z), dir * MAXD, params)
            if not (high and high.Instance.CanCollide) then return nil end
            return low.Distance
        end

        for dx = -range, range, step do
            for dz = -range, range, step do
                local x, z = center.X + dx, center.Z + dz
                local down = workspace:Raycast(Vector3.new(x, center.Y + 6, z), Vector3.new(0, -80, 0), params)
                if down and down.Instance.CanCollide then
                    local ground = down.Position.Y
                    local quick = 0
                    for _, d in ipairs(cardinals) do
                        if wallDist(x, ground, z, d) then quick += 1 end
                    end
                    local up = workspace:Raycast(Vector3.new(x, ground + 3, z), Vector3.new(0, 30, 0), params)
                    local headroomOk = not (up and up.Distance < 5)

                    if quick >= 3 and headroomOk then
                        local walls, minD = 0, math.huge
                        local dists = {}
                        for i, d in ipairs(dirs) do
                            local wd = wallDist(x, ground, z, d)
                            dists[i] = wd
                            if wd then
                                walls += 1
                                if wd < minD then minD = wd end
                            end
                        end

                        local ok = true
                        if minD < 4 then ok = false end
                        for i = 1, 4 do
                            local a, b = dists[i], dists[i + 4]
                            if a and b and (a + b) < 9 then ok = false end
                        end

                        if ok then
                            local score = walls * 10 + (up and 6 or 0) + math.min(minD, 8) * 0.5
                            local dist = math.sqrt(dx * dx + dz * dz)
                            if score > bestScore or (score == bestScore and dist < bestDist) then
                                best = Vector3.new(x, ground + 3, z)
                                bestScore, bestDist, bestWalls = score, dist, walls
                            end
                        end
                    end
                end
                n += 1
                if n % 80 == 0 then task.wait() end
            end
        end
        return best, bestWalls
    end

    local function trigger(h)
        local s = state[h]
        s.active = true
        s.done = false
        s.lastTrigger = tick()
        s.lastSeen = tick()
        s.started = tick()
        s.lastSearch = 0

        local hrp = getHRP()
        if not hrp then return end

        if h == "tsunami" then
            teleportToPlatform()
            s.done = true
            setStatus("Escape Tsunami -> Safe Platform")
        elseif h == "supernova" then
            s.before = hrp.CFrame
            task.spawn(function()
                local pos, score = findCover()
                local r = getHRP()
                if pos and r and s.active then
                    r.AssemblyLinearVelocity = Vector3.zero
                    r.CFrame = CFrame.new(pos)
                    setStatus("Escape Supernova -> cover (" .. tostring(score) .. "/8 walls)")
                else
                    setStatus("Escape Supernova: no cover found")
                end
                s.done = true
            end)
        elseif h == "nuke" then
            s.before = hrp.CFrame
            setStatus("Escape Nuke: searching shelter...")
        end
    end

    local function deactivate(h, delay)
        local s = state[h]
        s.active = false
        if delay <= 0 then return end

        if h == "tsunami" then
            returnToGround()
        else
            local hrp = getHRP()
            if hrp and s.before then
                hrp.AssemblyLinearVelocity = Vector3.zero
                hrp.CFrame = s.before
            end
        end
        s.before = nil
        setStatus("Escape: returned")
    end

    local function monitor()
        while true do
            local seen = scanTexts()
            local delay = getNum("escapeReturn", 8)
            for h, s in pairs(state) do
                if featureOn(h) then
                    if seen[h] then
                        s.lastSeen = tick()
                        if not s.active and tick() - s.lastTrigger > 4 then trigger(h) end
                    elseif s.active and tick() - s.lastSeen >= math.max(delay, 1) then
                        deactivate(h, delay)
                    end

                    if h == "nuke" and s.active and not s.done then
                        if tick() - s.started > 12 then
                            s.done = true
                            setStatus("Escape Nuke: shelter not found")
                        elseif tick() - s.lastSearch >= 0.3 then
                            s.lastSearch = tick()
                            local pos, exact = findShelter()
                            local hrp = getHRP()
                            if pos and hrp then
                                hrp.AssemblyLinearVelocity = Vector3.zero
                                hrp.CFrame = CFrame.new(pos)
                                s.done = true
                                setStatus("Escape Nuke -> Shelter")
                            end
                        end
                    end
                end
            end
            task.wait(0.2)
        end
    end

    local function anyOn()
        return featureOn("nuke") or featureOn("tsunami") or featureOn("supernova")
    end

    local function ensureMonitor()
        if not monThread then monThread = task.spawn(monitor) end
    end

    local function maybeStop()
        if not anyOn() and monThread then
            pcall(task.cancel, monThread)
            monThread = nil
        end
    end

    local function resetState(h)
        local s = state[h]
        s.active = false
        s.done = false
        s.before = nil
    end

    addToggle("escNuke", "Escape Nuke", function()
        shelterConn = workspace.DescendantAdded:Connect(function(d)
            if (d:IsA("Model") or d:IsA("BasePart")) and hasKey(d.Name:lower(), SHELTER_KEYS) then
                lastShelterAdded = tick()
            end
        end)
        ensureMonitor()
    end, function()
        if shelterConn then shelterConn:Disconnect() shelterConn = nil end
        resetState("nuke")
        maybeStop()
    end)

    addToggle("escTsunami", "Escape Tsunami", function()
        ensureMonitor()
    end, function()
        resetState("tsunami")
        maybeStop()
    end)

    addToggle("escSupernova", "Escape Supernova", function()
        ensureMonitor()
    end, function()
        resetState("supernova")
        maybeStop()
    end)

    addNumberRow("Return Delay (s, 0 = stay)", "escapeReturn", "8", "8")

    end

    local function restoreAll()
        for inst, en in pairs(touchedEffects) do
            if inst.Parent then pcall(function() inst.Enabled = en end) end
            touchedEffects[inst] = nil
        end
        for a, v in pairs(touchedAtmo) do
            if a.Parent then pcall(function() a.Density = v.Density a.Haze = v.Haze end) end
            touchedAtmo[a] = nil
        end
        for obj, props in pairs(touchedGui) do
            if obj.Parent then
                for p, v in pairs(props) do pcall(function() obj[p] = v end) end
            end
            touchedGui[obj] = nil
        end
        if fogOrig and fogOrig.End >= 5000 then
            pcall(function()

                Lighting.FogEnd = fogOrig.End
            end)
        end
    end

    local function remember(obj, prop)
        local t = touchedGui[obj]
        if not t then t = {} touchedGui[obj] = t end
        if t[prop] == nil then t[prop] = obj[prop] end
    end

    local function getControls()
        if controls then return controls end
        pcall(function()
            local ps = LocalPlayer:FindFirstChild("PlayerScripts")
            local pm = ps and ps:FindFirstChild("PlayerModule")
            if pm then controls = require(pm):GetControls() end
        end)
        return controls
    end

    local function realMove()
        local c = getControls()
        if c then
            local ok, v = pcall(function() return c:GetMoveVector() end)
            if ok and typeof(v) == "Vector3" then return v end
        end
        if UserInputService:GetFocusedTextBox() then return Vector3.zero end
        local x = (UserInputService:IsKeyDown(Enum.KeyCode.D) and 1 or 0) - (UserInputService:IsKeyDown(Enum.KeyCode.A) and 1 or 0)
        local z = (UserInputService:IsKeyDown(Enum.KeyCode.S) and 1 or 0) - (UserInputService:IsKeyDown(Enum.KeyCode.W) and 1 or 0)
        return Vector3.new(x, 0, z)
    end

    local function fixMovement()
        if not (cleanOpt.reversed or cleanOpt.forced) then return end
        if features.fly and features.fly.active then return end
        local hum = getHum()
        if not hum then return end

        if cleanOpt.forced then
            hum.AutoRotate = true
            local hrp = getHRP()
            if hrp then
                hrp.AssemblyAngularVelocity = Vector3.zero
                for _, o in ipairs(hrp:GetChildren()) do
                    if (o:IsA("BodyAngularVelocity") or o:IsA("AngularVelocity") or o:IsA("BodyGyro"))
                        and o.Name:sub(1, 6) ~= "ProFly" then
                        o:Destroy()
                    end
                end
            end
        end
        if cleanOpt.reversed and hum.WalkSpeed < 0 then hum.WalkSpeed = 16 end

        hum:Move(realMove(), true)
    end

    local function firstStep()
        if cleanOpt.shake then
            local hum = getHum()
            if hum and hum.CameraOffset ~= Vector3.zero then hum.CameraOffset = Vector3.zero end
        end
    end

    local function camStep()
        if cleanOpt.shake then
            local cam = workspace.CurrentCamera
            if cam then savedCF = cam.CFrame end
        end
    end

    local function lateStep()
        local cam = workspace.CurrentCamera
        if cam then
            if cleanOpt.shake and savedCF then cam.CFrame = savedCF end

            if cleanOpt.flipped or cleanOpt.dizzy or cleanOpt.crazy then
                local cf = cam.CFrame
                local look = cf.LookVector
                if math.abs(look.Y) < 0.995 then
                    cam.CFrame = CFrame.lookAt(cf.Position, cf.Position + look)
                end
            end

            if (cleanOpt.shake or cleanOpt.crazy or cleanOpt.dizzy) and cam.FieldOfView ~= baseFov then
                cam.FieldOfView = baseFov
            end
        end

        if cleanOpt.fog then
            if Lighting.FogEnd < 100000 then Lighting.FogEnd = 100000 end
            if Lighting.FogStart ~= 0 then Lighting.FogStart = 0 end
        end
        if cleanOpt.blind then
            pcall(function() RunService:Set3dRenderingEnabled(true) end)
        end

        fixMovement()
    end

    local function nameHint(n)
        n = n:lower()
        for _, h in ipairs({ "blind", "black", "dark", "vignette", "overlay", "fade", "cover", "blur" }) do
            if n:find(h, 1, true) then return true end
        end
        return false
    end

    local function isDark(c)
        return (c.R + c.G + c.B) < 0.45

    local function scanPost(root)
        for _, inst in ipairs(root:GetChildren()) do
            local kill =
                ((cleanOpt.blind or cleanOpt.dizzy or cleanOpt.crazy) and (inst:IsA("BlurEffect") or inst:IsA("DepthOfFieldEffect")))
                or ((cleanOpt.blind or cleanOpt.crazy) and inst:IsA("ColorCorrectionEffect"))
            if kill and inst.Enabled then
                if touchedEffects[inst] == nil then touchedEffects[inst] = true end
                inst.Enabled = false
            elseif cleanOpt.fog and inst:IsA("Atmosphere") then
                if not touchedAtmo[inst] then
                    touchedAtmo[inst] = { Density = inst.Density, Haze = inst.Haze }
                end
                inst.Density = 0
                inst.Haze = 0
            end
        end
    end

    local function scanGui()
        local cam = workspace.CurrentCamera
        local vp = cam and cam.ViewportSize or Vector2.new(1920, 1080)
        for _, obj in ipairs(PlayerGui:GetDescendants()) do
            if obj:IsA("GuiObject") and obj.Visible and not isOwnGui(obj) then
                local sz = obj.AbsoluteSize
                local big = sz.X >= vp.X * 0.9 and sz.Y >= vp.Y * 0.9

                if big and (cleanOpt.flipped or cleanOpt.dizzy or cleanOpt.crazy)
                    and obj.Rotation ~= 0 and obj.Parent:IsA("ScreenGui") then
                    remember(obj, "Rotation")
                    obj.Rotation = 0
                end

                if big and cleanOpt.blind
                    and not obj:FindFirstChildWhichIsA("GuiButton", true)
                    and not obj:FindFirstChildWhichIsA("TextBox", true) then
                    if obj:IsA("Frame") and obj.BackgroundTransparency < 0.9
                        and (isDark(obj.BackgroundColor3) or nameHint(obj.Name)) then
                        remember(obj, "BackgroundTransparency")
                        obj.BackgroundTransparency = 1
                    elseif obj:IsA("ImageLabel") and obj.ImageTransparency < 0.9 and nameHint(obj.Name) then
                        remember(obj, "ImageTransparency")
                        obj.ImageTransparency = 1
                    end
                end
            end
        end
    end

        if workspace.CurrentCamera then scanPost(workspace.CurrentCamera) end

            end)
        end

oldNamecall = hookmetamethod(game, "__namecall", newcclosure(function(self, ...)
    local method = getnamecallmethod()
    local args = {...}
    
    if silentAimEnabled and silentTargetPart and (method == "FireServer" or method == "InvokeServer") then
        -- Cari argumen yang isinya posisi (Vector3) atau CFrame (tembakan)
        for i, arg in ipairs(args) do
            if typeof(arg) == "Vector3" then
                args[i] = silentTargetPart.Position
            elseif typeof(arg) == "CFrame" then
                args[i] = CFrame.new(silentTargetPart.Position)
            elseif typeof(arg) == "Ray" then
                args[i] = Ray.new(arg.Origin, (silentTargetPart.Position - arg.Origin).Unit * 1000)
            end
        end
        return oldNamecall(self, unpack(args))
    end
    
    return oldNamecall(self, ...)
end))

        if bind == "Mobile" then return mobileAimActive end
        if bind:sub(1, 11) == "MouseButton" then
            local ok, ut = pcall(function() return Enum.UserInputType[bind] end)
            return (ok and ut and UserInputService:IsMouseButtonPressed(ut)) or false
        end
        local ok, kc = pcall(function() return Enum.KeyCode[bind] end)
        return (ok and kc and UserInputService:IsKeyDown(kc)) or false

    local function isAiming()
        if UserInputService:GetFocusedTextBox() then return false end
        return isBindDown(aimBind)

    local function getAimPart(char)
        local mode = settings.values.aimPart or "Head"
        if mode == "Torso" then
            return char:FindFirstChild("UpperTorso")
                or char:FindFirstChild("Torso")
                or char:FindFirstChild("HumanoidRootPart")
        end
        return char:FindFirstChild("Head") or char:FindFirstChild("HumanoidRootPart")
    end

    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    local function isVisible(cam, part, char)
        local filter = { char }
        if LocalPlayer.Character then table.insert(filter, LocalPlayer.Character) end
        rayParams.FilterDescendantsInstances = filter
        local origin = cam.CFrame.Position
        return workspace:Raycast(origin, part.Position - origin, rayParams) == nil
    end

    local function evaluate(player, cam, center, useFov)
        if player == LocalPlayer then return end
        local char = player.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then return end
        local part = getAimPart(char)
        if not part then return end

        local sp, onScreen = cam:WorldToViewportPoint(part.Position)
        if not onScreen then return end
        local sd = (Vector2.new(sp.X, sp.Y) - center).Magnitude
        if useFov and sd > getNum("aimFov", 200) then return end

        if aimOpt.wall and not isVisible(cam, part, char) then return end
        return part, sd
    end

    local function findTarget(cam, center)
        local best, bestScore
        for _, player in ipairs(Players:GetPlayers()) do
            local part, sd = evaluate(player, cam, center, true)
            if part and (not bestScore or sd < bestScore) then
                best, bestScore = player, sd
            end
        end
        return best
    end

    --==========================================================--
    --  AIM DÖNGÜSÜ — PC + MOBİL ORTAK
    --==========================================================--
    local function aimStep(dt)
        local cam = workspace.CurrentCamera
        if not cam then return end
        local center = cam.ViewportSize / 2

        local fov = getNum("aimFov", 200)
        fovCircle.Position = UDim2.fromOffset(center.X, center.Y)
        fovCircle.Size = UDim2.fromOffset(fov * 2, fov * 2)
        fovCircle.Visible = aimOpt.fovCircle == true

        -- Mobilde buton görünürlüğünü ayarla
        if mobileAimBtn then
            mobileAimBtn.Visible = aimOpt.aimbot ~= false
        end

        if not isAiming() then
    lockedTarget = nil
    lastLockName = nil
    silentAimEnabled = false
    silentTargetPart = nil
    return
end

        local part
        if lockedTarget then
            part = evaluate(lockedTarget, cam, center, false)
            if not part then lockedTarget = nil end
        end

        if not lockedTarget then
            lockedTarget = findTarget(cam, center)
            if lockedTarget then
                part = evaluate(lockedTarget, cam, center, false)
            end
        end
        if not lockedTarget or not part then
            lastLockName = nil
            return
        end

        if lastLockName ~= lockedTarget.Name then
            lastLockName = lockedTarget.Name
            setStatus("Aimbot -> " .. lockedTarget.Name)
        end

        local s = math.max(getNum("aimSmooth", 2), 1)
        local alpha = (s <= 1) and 1 or (1 - (1 - 1 / s) ^ (dt * 60))
        local camPos = cam.CFrame.Position
            local camPos = cam.CFrame.Position
    cam.CFrame = cam.CFrame:Lerp(CFrame.lookAt(camPos, part.Position), alpha)
    
    -- TAMBAHAN SILENT AIM
    if aimOpt.silent and part then
        silentAimEnabled = true
        silentTargetPart = part
    else
        silentAimEnabled = false
        silentTargetPart = nil
    end

    if aimOpt.silent then
        return
    end
end  <--- Ini baris 3138 (end penutup aimStep)

                capturing = true
                bindBtn.Text = "Press a key..."
            end))

                if not capturing then return end
                local name
                if input.UserInputType == Enum.UserInputType.Keyboard then
                    if input.KeyCode == Enum.KeyCode.Escape then
                        capturing = false
                        bindBtn.Text = prettyBind(aimBind)
                        return
                    end
                    name = input.KeyCode.Name
                elseif input.UserInputType == Enum.UserInputType.MouseButton1
                    or input.UserInputType == Enum.UserInputType.MouseButton2
                    or input.UserInputType == Enum.UserInputType.MouseButton3 then
                    name = input.UserInputType.Name
                end
                if name and name ~= "Unknown" then
                    capturing = false
                    aimBind = name
                    settings.values.aimKey = name
                    saveSettings()
                    bindBtn.Text = prettyBind(name)
                    setStatus("Aim Bind: " .. prettyBind(name))
                end
            end))
        end
    end

    end

    addAimOpt("fovCircle", "Show FOV Circle", true)
    addAimOpt("silent", "Silent Aim", false)
end

local function revList(t)
    local out = {}
    for i, v in ipairs(t) do out[i] = v:reverse() end
    return out
end

local function revMap(t)
    local out = {}
    for k, v in pairs(t) do
        out[k:reverse()] = v:reverse()
    end
    return out
end

local COMMAND_DESC = "Meta Word: must be unlocked in-game first."

local function cleanWord(s)
    return (s:gsub("[%s_']", ""))
end
local function normKey(s)
    return cleanWord(s):lower()

local WORD_GROUPS = {}
do
    local seen = {}
    for _, g in ipairs(RAW_GROUPS) do
        local described, plain = {}, {}
        for _, raw in ipairs(g.words) do
            local w = cleanWord(raw)
            local key = normKey(raw)
            if w ~= "" and not seen[key] then
                seen[key] = true
                if g.name == "Commands" and not DESCS[key] then DESCS[key] = COMMAND_DESC end
                table.insert(DESCS[key] and described or plain, w)
            end
        end
        local words = {}
        for _, w in ipairs(described) do table.insert(words, w) end
        for _, w in ipairs(plain) do table.insert(words, w) end
        table.insert(WORD_GROUPS, { name = g.name, words = words })
    end
end

local function isOwnGui(obj)
    local sg = obj:FindFirstAncestorOfClass("ScreenGui")
    return sg ~= nil and sg.Name:sub(1, 3) == "Pro"
end

local function isReallyVisible(gui)
    local cur = gui
    while cur and cur ~= game do
        if cur:IsA("GuiObject") and not cur.Visible then return false end
        if cur:IsA("ScreenGui") and not cur.Enabled then return false end
        cur = cur.Parent
    end
    return true
end

local function getWordToSend(random)
    if random then
        local group = groupByName[selectedGroup]
        if group and group.words and #group.words > 0 then
            local rw = group.words[math.random(1, #group.words)]
            if boxes.word then boxes.word.Text = rw end
            settings.values.word = rw
            saveSettings()
            return rw
        end
    end
    return boxes.word and boxes.word.Text or ""
end

local function fireSig(signal, ...)
    local args = { ... }
    if firesignal then
        pcall(function() firesignal(signal, table.unpack(args)) end)
    elseif getconnections then
        pcall(function()
            for _, c in ipairs(getconnections(signal)) do c:Fire(table.unpack(args)) end
        end)
    end
end

local SEND_HINTS = { "send", "submit", "confirm", "write", "gönder", "gonder", "yolla", "tamam" }
local function findSendButtons(box)
    local found = {}
    local container = box.Parent
    for _ = 1, 3 do
        if not container or container:IsA("ScreenGui") then break end
        for _, d in ipairs(container:GetDescendants()) do
            if d:IsA("GuiButton") and isReallyVisible(d) and not isOwnGui(d) then
                local n = (d.Name .. " " .. (d:IsA("TextButton") and d.Text or "")):lower()
                for _, h in ipairs(SEND_HINTS) do
                    if n:find(h, 1, true) then
                        table.insert(found, d)
                        break
                    end
                end
            end
        end
        if #found > 0 then break end
        container = container.Parent
    end
    return found
end

local function clickButton(btn)
    fireSig(btn.MouseButton1Down)
    fireSig(btn.MouseButton1Up)
    fireSig(btn.MouseButton1Click)
    fireSig(btn.Activated)
    if VIM then
        pcall(function()
            local inset = GuiService:GetGuiInset()
            local p = btn.AbsolutePosition + btn.AbsoluteSize / 2 + inset
            VIM:SendMouseButtonEvent(p.X, p.Y, 0, true, game, 0)
            task.wait()
            VIM:SendMouseButtonEvent(p.X, p.Y, 0, false, game, 0)
        end)
    end
end

local function submitIntoBox(box, word)
    box.Text = word
    box:CaptureFocus()
    task.wait(0.05)
    if box.Text ~= word then box.Text = word end

    local function submitted()
        return box.Text ~= word or not isReallyVisible(box) or not box:IsFocused()
    end

    if VIM then
        pcall(function()
            VIM:SendKeyEvent(true, Enum.KeyCode.Return, false, game)
            task.wait()
            VIM:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
        end)
    elseif keypress and keyrelease then
        pcall(function()
            keypress(0x0D)
            task.wait()
            keyrelease(0x0D)
        end)
    end
    task.wait(0.05)
    if submitted() then return end

    fireSig(box.FocusLost, true)
    pcall(function() box:ReleaseFocus(true) end)
    task.wait(0.05)
    if box.Text ~= word or not isReallyVisible(box) then return end

    for _, btn in ipairs(findSendButtons(box)) do
        clickButton(btn)
    end
end

local function executeWordSubmit(word)
    if not word or word == "" then return false end

    local r = findWordRemote()
    if r then
        local wr = ReplicatedStorage:FindFirstChild("WriterRemotes")
        local typing = wr and wr:FindFirstChild("WriterTyping")
        if typing then pcall(function() typing:FireServer(word) end) end
        local ok = pcall(function() r:FireServer(word) end)
        if ok then
            lastSendInfo = " [remote: " .. r.Name .. "]"
            return true
        end
    end

    lastSendInfo = " [UI]"
    local playerGui = LocalPlayer:FindFirstChild("PlayerGui")
    if playerGui then
        task.spawn(function()
            local targets, visibleOnes = {}, {}
            for _, box in ipairs(playerGui:GetDescendants()) do
                if box:IsA("TextBox") and not isOwnGui(box) then
                    table.insert(targets, box)
                    if isReallyVisible(box) then table.insert(visibleOnes, box) end
                end
            end
            for _, box in ipairs(#visibleOnes > 0 and visibleOnes or targets) do
                local ok, err = pcall(submitIntoBox, box, word)
                if not ok then warn("[ProSurvive] submit: " .. tostring(err)) end
            end
        end)
    end
    return true
end

local smartConnections = {}
local senderMode = nil
local lastFiredTick = 0
local COOLDOWN = 3.5

local exactPatterns = {
    "your turn to write",
    "your turn",
    "it's your turn",
    "sıra sende",
    "sira sende"
}

local function tryInstantFire(sourceName)
    local now = tick()
    if (now - lastFiredTick) < COOLDOWN then return end

    local word = getWordToSend(senderMode == "random")
    if word == "" then return end

    lastFiredTick = now
    local success = executeWordSubmit(word)
    if success then
        setStatus("⚡ Sent (" .. sourceName .. ") -> " .. word .. lastSendInfo)
    end
end

local function isMatch(text)
    if type(text) ~= "string" or text == "" then return false end
    local lower = text:lower()
    for _, p in ipairs(exactPatterns) do
        if lower:find(p, 1, true) then return true end
    end
    return false
end

local function inspectElement(element)
    if element:IsA("TextBox") and not isOwnGui(element) then
        local function onBoxVisible()
            if isReallyVisible(element) and (tick() - lastFiredTick) >= COOLDOWN then
                tryInstantFire("Box-Opened")
            end
        end
        table.insert(smartConnections, element:GetPropertyChangedSignal("Visible"):Connect(onBoxVisible))
        onBoxVisible()
        return
    end

    if not (element:IsA("TextLabel") or element:IsA("TextButton")) then return end
    if isOwnGui(element) then return end

    local function check()
        if not isReallyVisible(element) then return end

        if isMatch(element.Text) then
            tryInstantFire("Turn-Text")
            return
        end

        local txt = element.Text:lower()
        if (txt == "write" or txt:find("^write%s*")) and element.AbsoluteSize.Y > 20 then
            tryInstantFire("Write-Button")
        end
    end

    table.insert(smartConnections, element:GetPropertyChangedSignal("Text"):Connect(check))
    table.insert(smartConnections, element:GetPropertyChangedSignal("Visible"):Connect(check))
    table.insert(smartConnections, element:GetPropertyChangedSignal("Parent"):Connect(check))

    check()
end

local pollThread
local function startSender()
    if pollThread then return end
    local playerGui = LocalPlayer:WaitForChild("PlayerGui")

    for _, desc in ipairs(playerGui:GetDescendants()) do
        inspectElement(desc)
    end
    table.insert(smartConnections, playerGui.DescendantAdded:Connect(inspectElement))

    local wr = ReplicatedStorage:FindFirstChild("WriterRemotes")
    local prompt = wr and wr:FindFirstChild("WriterPrompt")
    local chosenEv = wr and wr:FindFirstChild("WriterChosen")
    local announcedEv = wr and wr:FindFirstChild("WordAnnounced")
    local finalDuelEv = wr and wr:FindFirstChild("FinalDuel")
    local chosenName, chosenAt, duelAt = nil, 0, 0
    local inFinalDuel = false

    if chosenEv then
        table.insert(smartConnections, chosenEv.OnClientEvent:Connect(function(name)
            chosenName, chosenAt = name, tick()
        end))
    end

    -- YENİ: FinalDuel Desteği
    if finalDuelEv then
        table.insert(smartConnections, finalDuelEv.OnClientEvent:Connect(function(data)
            inFinalDuel = true
            task.delay(math.random(30, 60) / 100, function()
                if senderMode then
                    tryInstantFire("FinalDuel")
                end
            end)
        end))
    end

    local rouletteEv = wr and wr:FindFirstChild("WriterRoulette")
    if rouletteEv then
        table.insert(smartConnections, rouletteEv.OnClientEvent:Connect(function(mode, data)
            if mode == "duel" and type(data) == "table" then
                for _, p in ipairs(data) do
                    if type(p) == "table" and p.name == LocalPlayer.Name then
                        chosenName, chosenAt, duelAt = LocalPlayer.Name, tick(), tick()
                    end
                end
            end
        end))
    end

    if prompt then
        table.insert(smartConnections, prompt.OnClientEvent:Connect(function()
            task.delay(math.random(30, 80) / 100, function()
                if not senderMode then return end
                if inFinalDuel or (duelAt > 0 and (tick() - duelAt < 15)) then
                    tryInstantFire("Prompt-Duel")
                    return
                end
                if chosenName and chosenName ~= LocalPlayer.Name and tick() - chosenAt < 10 then
                    return
                end
                tryInstantFire("Prompt")
            end)
        end))
    end

    if announcedEv then
        table.insert(smartConnections, announcedEv.OnClientEvent:Connect(function(text)
            inFinalDuel = false
            if type(text) ~= "string" then return end
            local clean = text:gsub("<[^>]+>", "")
            if clean:sub(1, #LocalPlayer.Name + 6) == LocalPlayer.Name .. " said " then
                setStatus("Accepted: " .. clean:sub(1, 60))
            end
        end))
    end

    pollThread = task.spawn(function()
        local boxWasOpen, textWasMatch = false, false
        while true do
            local boxOpen, textMatch = false, false
            for _, d in ipairs(playerGui:GetDescendants()) do
                if not isOwnGui(d) then
                    if d:IsA("TextBox") then
                        if isReallyVisible(d) then boxOpen = true end
                    elseif (d:IsA("TextLabel") or d:IsA("TextButton")) and isMatch(d.Text) and isReallyVisible(d) then
                        textMatch = true
                    end
                end
            end
            if boxOpen and not boxWasOpen then tryInstantFire("Box-Poll") end
            if textMatch and not textWasMatch then tryInstantFire("Text-Poll") end
            boxWasOpen, textWasMatch = boxOpen, textMatch
            task.wait(0.2)
        end
    end)
end

local function stopSender()
    if pollThread then pcall(task.cancel, pollThread) pollThread = nil end
    for _, conn in ipairs(smartConnections) do
        if conn and conn.Disconnect then
            pcall(function() conn:Disconnect() end)
        end
    end
    smartConnections = {}
end

do
    local SLOTS_FILE = "ProSurviveSlots.json"
    local SLOT_COUNT = 5
    local EXCLUDE = { menuKey = true, aimKey = true, lastSlot = true }
    local slots = {}
    local selected = tonumber(settings.values.lastSlot) or 1
    if selected < 1 or selected > SLOT_COUNT then selected = 1 end

    local function readSlots()
        if not (isfile and readfile) then return end
        local ok, data = pcall(function()
            if isfile(SLOTS_FILE) then
                return HttpService:JSONDecode(readfile(SLOTS_FILE))
            end
        end)
        if ok and type(data) == "table" then slots = data end
    end

    local function writeSlots()
        if not writefile then return false end
        return (pcall(function()
            writefile(SLOTS_FILE, HttpService:JSONEncode(slots))
        end))
    end

    readSlots()

    local function slotName(i)
        local s = slots[tostring(i)]
        return s and s.name or nil
    end

    -- Dropdown başlığı: isim yazılabilir kutu + ok
    local row = newRow()
    local nameBox = Instance.new("TextBox")
    nameBox.Parent = row
    nameBox.BackgroundColor3 = C.input

    nameBox.Size = UDim2.new(1, -52, 0, 24)
    nameBox.ClearTextOnFocus = false
    nameBox.Font = Enum.Font.GothamBold
    nameBox.PlaceholderText = "Type a name..."
    nameBox.PlaceholderColor3 = C.sub
    nameBox.Text = slotName(selected) or ""
    nameBox.TextColor3 = C.text
    nameBox.TextSize = 12
    nameBox.TextXAlignment = Enum.TextXAlignment.Left

    arrow.Text = "▼"
    arrow.TextColor3 = Color3.fromRGB(255, 255, 255)
    arrow.TextSize = 10
    corner(arrow, 6)
    onAccent(function(c) arrow.BackgroundColor3 = c end)

    rowOrder += 1
    local panel = Instance.new("Frame")
    panel.Parent = curPage
    panel.BackgroundColor3 = C.title
    panel.BorderSizePixel = 0
    panel.LayoutOrder = rowOrder
    panel.Size = UDim2.new(1, 0, 0, SLOT_COUNT * 30 + (SLOT_COUNT - 1) * 4 + 12)

        arrow.Text = open and "▲" or "▼"

        for _, ch in ipairs(panel:GetChildren()) do
            if ch:IsA("TextButton") then ch:Destroy() end

        end

    end))

    local function doSave()
        local name = nameBox.Text:gsub("^%s+", ""):gsub("%s+$", "")
        name = name:sub(1, 24)
        if name == "" then name = "Slot " .. selected end

        -- 1) Aynı isimli slot varsa onun üstüne yaz
        local target
        for i = 1, SLOT_COUNT do
            local nm = slotName(i)
            if nm and nm:lower() == name:lower() then target = i break end
        end
        -- 2) Seçili slot boşsa oraya
        if not target and not slotName(selected) then target = selected end
        -- 3) Yoksa ilk boş slota (dolu bir slotun yanlışlıkla ezilmesini önler)
        if not target then
            for i = 1, SLOT_COUNT do
                if not slotName(i) then target = i break end
            end
        end
        -- 4) Hepsi doluysa seçili slotun üstüne yaz
        target = target or selected

        local values, toggles = {}, {}
        for k, v in pairs(settings.values) do
            if not EXCLUDE[k] then values[k] = v end
        end
        for k, v in pairs(settings.toggles) do toggles[k] = v end

        slots[tostring(target)] = { name = name, values = values, toggles = toggles }
        selected = target
        settings.values.lastSlot = target
        saveSettings()
        nameBox.Text = name
        refreshList()
        if writeSlots() then
            setStatus("Saved -> " .. name .. " (Slot " .. target .. ")")
        else
            setStatus("Error: Save failed")
        end
    end

    local function doLoad()
        local slot = slots[tostring(selected)]
        if not slot then
            setStatus("Slot " .. selected .. " is Empty")
            return
        end

        for k, v in pairs(slot.values or {}) do
            if k == "theme" then
                for _, t in ipairs(THEMES) do
                    if t.name == v then applyAccent(t.color) end
                end
                settings.values.theme = v
            elseif k == "wordGroup" then
                if groupByName[v] then
                    selectedGroup = v
                    settings.values.wordGroup = v
                    groupVal.Text = groupLabel(v)
                end
            elseif boxes[k] then
                boxes[k].Text = tostring(v)
                settings.values[k] = tostring(v)
            elseif cycleSetters[k] then
                cycleSetters[k](v)
            elseif applyHooks[k] then
                applyHooks[k](v)
            elseif not EXCLUDE[k] then
                settings.values[k] = v
            end
        end

        for key, feature in pairs(features) do
            feature.set((slot.toggles or {})[key] == true)
        end

        saveSettings()
        setStatus("Loaded -> " .. (slot.name or ("Slot " .. selected)))
    end

    local function doDelete()
        if slots[tostring(selected)] then
            slots[tostring(selected)] = nil
            writeSlots()
            nameBox.Text = ""
            refreshList()
            setStatus("Slot " .. selected .. " deleted")
        else
            setStatus("Slot " .. selected .. " is already Empty")
        end
    end

    addActionRow("Save Settings", "accent", doSave)
    addActionRow("Load Settings", "accent", doLoad)
    addActionRow("Delete Slot", "danger", doDelete)
    addInfo("Pick a slot from the dropdown, or type a name and press Save. Empty slots show 'Empty'.", 30)
end

local function rejoin()
    setStatus("Rejoining...")
    task.spawn(function()
        local ok, err = pcall(function()
            if #Players:GetPlayers() <= 1 then
                LocalPlayer:Kick("\nRejoining...")
                task.wait()
                TeleportService:Teleport(game.PlaceId, LocalPlayer)
            else
                TeleportService:TeleportToPlaceInstance(game.PlaceId, game.JobId, LocalPlayer)
            end
        end)
        if not ok then
            warn("[ProSurvive] Rejoin: " .. tostring(err))
            setStatus("Error: Rejoin failed")
        end
    end)
end

local hopping = false
local function serverHop()
    if hopping then return end
    hopping = true
    setStatus("Searching servers...")

    task.spawn(function()
        local found, cursor, failed = nil, "", false

        for _ = 1, 5 do
            local url = "https://games.roblox.com/v1/games/" .. tostring(game.PlaceId)
                .. "/servers/Public?sortOrder=Asc&limit=100"
            if cursor ~= "" then
                url = url .. "&cursor=" .. HttpService:UrlEncode(cursor)
            end

            local ok, res = pcall(function()
                return HttpService:JSONDecode(game:HttpGet(url))
            end)
            if not ok or type(res) ~= "table" or type(res.data) ~= "table" then
                failed = true
                break
            end

            local candidates = {}
            for _, s in ipairs(res.data) do
                if s.id ~= game.JobId and s.playing and s.maxPlayers and s.playing < s.maxPlayers - 1 then
                    table.insert(candidates, s)
                end
            end
            if #candidates > 0 then
                found = candidates[math.random(#candidates)]
                break
            end

            cursor = res.nextPageCursor or ""
            if cursor == "" then break end
        end

        if found then
            setStatus("Hopping to new server...")
            local ok, err = pcall(function()
                TeleportService:TeleportToPlaceInstance(game.PlaceId, found.id, LocalPlayer)
            end)
            if not ok then
                warn("[ProSurvive] ServerHop: " .. tostring(err))
                setStatus("Error: Hop failed")
            end
        else
            setStatus(failed and "Error: Server list failed" or "No server found")
        end

        task.delay(3, function() hopping = false end)
    end)
end
